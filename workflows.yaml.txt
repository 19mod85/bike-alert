name: Canyon Checker

on:
  schedule:
    - cron: '0 */3 * * *'  # Uruchamia automatycznie co 3 godziny
  workflow_dispatch:        # Umożliwia kliknięcie "Run" ręcznie z telefonu w każdej chwili

jobs:
  run-script:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: '3.10'

      - name: Install dependencies
        uses: py-actions/pip-dependency-install@v4
        with:
          subpackages: "requests beautifulsoup4"

      - name: Run checker
        env:
          NTFY_TOPIC: "tutaj_wpisz_nazwa_swojego_tematu_z_kroku_1"
        run: |
          python checker.py
